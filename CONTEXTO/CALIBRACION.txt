import json
import math
import os
import sys
import time

import numpy as np
import serial


# ============================================================================422
# PARAMETROS BASE
# ============================================================================
# VALORES CORRECTOS — reemplaza en ambos archivos
L1          = 150.0  # mm
L2          = 150.0
L2_EFECTIVO = 159.94   # sqrt(150² + 55.5²)
GAMMA       = 20.30    # atan2(55.5, 150) en grados
PUERTO_ARDUINO = "COM3"
BAUDRATE = 115200
TIMEOUT_MOVIMIENTO = 15.0
RADIO_BASE_SEGURO = 60.0

DISENO_CENTRO_X = 200.0
DISENO_CENTRO_Y = 0.0
PERFIL_PATH = os.path.join(os.path.dirname(__file__), "calibracion_perfil.json")
STOP_REQUESTED = False


# ============================================================================
# PERFIL DE CALIBRACION
# ============================================================================
def perfil_default():
    return {
        "workspace": {
            "offset_x": 0.0,
            "offset_y": 0.0,
            "escala_x": 1.0,
            "escala_y": 1.0,
            "rotacion_grados": 0.0,
        },
        "angulos": {
            "q1_gain": 1.0,
            "q2_gain": 1.0,
            "q1_offset": 0.0,
            "q2_offset": 0.0,
        },
        "movimiento": {
            "pasos_dibujo": 20,
            "pasos_traslado": 2,
            "timeout_movimiento": TIMEOUT_MOVIMIENTO,
        },
        "lapiz": {
            "settle_down_s": 0.15,
            "settle_up_s": 0.1,
        },
    }


def cargar_perfil():
    perfil = perfil_default()
    if not os.path.exists(PERFIL_PATH):
        return perfil

    try:
        with open(PERFIL_PATH, "r", encoding="utf-8") as f:
            user = json.load(f)
        for bloque, data in perfil.items():
            if isinstance(user.get(bloque), dict):
                data.update(user[bloque])
    except Exception as e:
        print(f"[WARN] No se pudo cargar perfil: {e}")
    return perfil


def guardar_perfil(perfil):
    with open(PERFIL_PATH, "w", encoding="utf-8") as f:
        json.dump(perfil, f, indent=2, ensure_ascii=False)
    print(f"Perfil guardado en: {PERFIL_PATH}")


def restablecer_espacio_trabajo(perfil):
    perfil["workspace"]["offset_x"] = 0.0
    perfil["workspace"]["offset_y"] = 0.0
    perfil["workspace"]["escala_x"] = 1.0
    perfil["workspace"]["escala_y"] = 1.0
    perfil["workspace"]["rotacion_grados"] = 0.0
    print("[RESET] Espacio de trabajo restaurado a valores base.")


def solicitar_paro():
    global STOP_REQUESTED
    STOP_REQUESTED = True
    print("[STOP] Se solicitó paro seguro de la operación.")


def limpiar_paro():
    global STOP_REQUESTED
    STOP_REQUESTED = False


def paro_solicitado():
    return STOP_REQUESTED


# ============================================================================
# CINEMATICA
# ============================================================================
def punto_es_seguro(px, py):
    distancia = math.sqrt(px ** 2 + py ** 2)
    alcance_maximo = L1 + L2_EFECTIVO - 5.0
    return RADIO_BASE_SEGURO <= distancia <= alcance_maximo


def ik(px, py):
    h2 = px ** 2 + py ** 2
    cos_q2 = np.clip((h2 - L1 ** 2 - L2_EFECTIVO ** 2) / (2 * L1 * L2_EFECTIVO), -1.0, 1.0)
    q2_efectivo = math.degrees(math.acos(cos_q2))
    beta = math.atan2(py, px)
    alfa = math.atan2(L2_EFECTIVO * math.sin(math.radians(q2_efectivo)), L1 + L2_EFECTIVO * math.cos(math.radians(q2_efectivo)))
    q1 = math.degrees(beta - alfa)
    q2_motor = q2_efectivo - GAMMA
    return q1, q2_motor


def fk(q1, q2):
    q1r = math.radians(q1)
    q2r = math.radians(q2 + GAMMA)
    x_codo = L1 * math.cos(q1r)
    y_codo = L1 * math.sin(q1r)
    x_tip = x_codo + L2_EFECTIVO * math.cos(q1r + q2r)
    y_tip = y_codo + L2_EFECTIVO * math.sin(q1r + q2r)
    return x_tip, y_tip


# ============================================================================
# APLICAR PERFIL
# ============================================================================
def aplicar_calibracion_xy(x, y, perfil):
    ws = perfil["workspace"]
    dx = x - DISENO_CENTRO_X
    dy = y - DISENO_CENTRO_Y

    dx *= ws["escala_x"]
    dy *= ws["escala_y"]

    ang = math.radians(ws["rotacion_grados"])
    dxr = dx * math.cos(ang) - dy * math.sin(ang)
    dyr = dx * math.sin(ang) + dy * math.cos(ang)

    return (
        DISENO_CENTRO_X + dxr + ws["offset_x"],
        DISENO_CENTRO_Y + dyr + ws["offset_y"],
    )


def aplicar_calibracion_angulos(q1, q2, perfil):
    ang = perfil["angulos"]
    q1c = q1 * ang["q1_gain"] + ang["q1_offset"]
    q2c = q2 * ang["q2_gain"] + ang["q2_offset"]
    return q1c, q2c


# ============================================================================
# SERIAL
# ============================================================================
def enviar_comando(arduino, q1, q2, z, timeout_s):
    if paro_solicitado():
        return False

    payload = f"{q1:.2f},{q2:.2f},{int(z)}\n"
    arduino.write(payload.encode("utf-8"))

    t0 = time.time()
    while True:
        if arduino.in_waiting > 0:
            resp = arduino.readline().decode("utf-8", errors="ignore").strip()
            if resp == "OK":
                return True
        if time.time() - t0 > timeout_s:
            return False
        if paro_solicitado():
            return False


def ir_home(arduino, timeout_s):
    print("Moviendo a home...")
    ok = enviar_comando(arduino, 0.0, 0.0, 0, timeout_s)
    if not ok:
        print("[WARN] Timeout yendo a home")


# ============================================================================
# TRAZO DE PRUEBA
# ============================================================================
def mover_xy(arduino, x, y, z, perfil):
    if paro_solicitado():
        return False

    if not punto_es_seguro(x, y):
        print(f"Punto inseguro: ({x:.1f}, {y:.1f})")
        return False

    q1, q2 = ik(x, y)
    q1, q2 = aplicar_calibracion_angulos(q1, q2, perfil)
    timeout_s = float(perfil["movimiento"]["timeout_movimiento"])
    return enviar_comando(arduino, q1, q2, z, timeout_s)


def trazar_polilinea(arduino, puntos_diseno, perfil):
    limpiar_paro()
    pasos_dibujo = max(3, int(perfil["movimiento"]["pasos_dibujo"]))
    pasos_traslado = max(2, int(perfil["movimiento"]["pasos_traslado"]))
    timeout_s = float(perfil["movimiento"]["timeout_movimiento"])

    # 1) Ir al inicio con lapiz arriba
    x0, y0 = aplicar_calibracion_xy(puntos_diseno[0][0], puntos_diseno[0][1], perfil)
    if not mover_xy(arduino, x0, y0, 0, perfil):
        return False

    # 2) Bajar lapiz
    q1, q2 = ik(x0, y0)
    q1, q2 = aplicar_calibracion_angulos(q1, q2, perfil)
    if not enviar_comando(arduino, q1, q2, 1, timeout_s):
        return False
    time.sleep(float(perfil["lapiz"]["settle_down_s"]))

    # 3) Dibujar segmentos
    for i in range(len(puntos_diseno) - 1):
        p1 = puntos_diseno[i]
        p2 = puntos_diseno[i + 1]
        for t in np.linspace(0, 1, pasos_dibujo):
            if paro_solicitado():
                print("[STOP] Operación detenida por el usuario.")
                return False
            xr = p1[0] + (p2[0] - p1[0]) * t
            yr = p1[1] + (p2[1] - p1[1]) * t
            x, y = aplicar_calibracion_xy(xr, yr, perfil)
            if not mover_xy(arduino, x, y, 1, perfil):
                return False

    # 4) Subir lapiz
    q1, q2 = ik(x0, y0)
    q1, q2 = aplicar_calibracion_angulos(q1, q2, perfil)
    if not enviar_comando(arduino, q1, q2, 0, timeout_s):
        return False
    time.sleep(float(perfil["lapiz"]["settle_up_s"]))

    # Pequenio traslado de confirmacion (evita marca por vibracion)
    if not mover_xy(arduino, x0, y0, 0, perfil):
        return False
    for _ in range(pasos_traslado - 1):
        if not mover_xy(arduino, x0, y0, 0, perfil):
            return False

    return True


def figura_cuadrado(lado=80.0):
    c = (DISENO_CENTRO_X, DISENO_CENTRO_Y)
    h = lado / 2.0
    return [
        (c[0] - h, c[1] - h),
        (c[0] + h, c[1] - h),
        (c[0] + h, c[1] + h),
        (c[0] - h, c[1] + h),
        (c[0] - h, c[1] - h),
    ]


def figura_circulo(radio=40.0, n=60):
    c = (DISENO_CENTRO_X, DISENO_CENTRO_Y)
    pts = []
    for a in np.linspace(0, 2 * np.pi, n):
        pts.append((c[0] + radio * np.cos(a), c[1] + radio * np.sin(a)))
    return pts

def modo_jog_centro(arduino, perfil):
    print("\nMODO JOG XY")
    print("Comandos: w/a/s/d mover XY, +/- cambiar paso, u lapiz arriba, j lapiz abajo")
    print("          h home, c guardar centro de hoja, x salir")

    q1 = 0.0
    q2 = 0.0
    x, y = fk(q1, q2)
    paso = 2.0
    z = 0

    ir_home(arduino, float(perfil["movimiento"]["timeout_movimiento"]))

    while True:
        cmd = input(f"[jog paso={paso:.1f}mm] > ").strip().lower()
        if cmd == "x":
            break
        elif cmd == "+":
            paso = min(20.0, paso + 1.0)
            continue
        elif cmd == "-":
            paso = max(0.5, paso - 0.5)
            continue
        elif cmd == "h":
            ir_home(arduino, float(perfil["movimiento"]["timeout_movimiento"]))
            q1, q2 = 0.0, 0.0
            x, y = fk(q1, q2)
            continue
        elif cmd == "u":
            z = 0
            if not enviar_comando(arduino, q1, q2, z, float(perfil["movimiento"]["timeout_movimiento"])):
                print("Timeout en comando")
            continue
        elif cmd == "j":
            z = 1
            if not enviar_comando(arduino, q1, q2, z, float(perfil["movimiento"]["timeout_movimiento"])):
                print("Timeout en comando")
            continue
        elif cmd == "c":
            print(f"Centro actual detectado en XY=({x:.2f}, {y:.2f})")
            confirmar = input("Guardar este punto como centro de hoja? (s/n): ").strip().lower()
            if confirmar == "s":
                perfil["workspace"]["offset_x"] = x - DISENO_CENTRO_X
                perfil["workspace"]["offset_y"] = y - DISENO_CENTRO_Y
                print("Centro de hoja guardado en offsets:")
                print(f"offset_x={perfil['workspace']['offset_x']:.2f}, offset_y={perfil['workspace']['offset_y']:.2f}")
            else:
                print("Centro no guardado.")
            continue

        nx, ny = x, y
        if cmd == "w":
            ny += paso
        elif cmd == "s":
            ny -= paso
        elif cmd == "a":
            nx -= paso
        elif cmd == "d":
            nx += paso
        else:
            print("Comando no valido")
            continue

        if not punto_es_seguro(nx, ny):
            print("Movimiento fuera de zona segura")
            continue

        nq1, nq2 = ik(nx, ny)
        cq1, cq2 = aplicar_calibracion_angulos(nq1, nq2, perfil)
        ok = enviar_comando(arduino, cq1, cq2, z, float(perfil["movimiento"]["timeout_movimiento"]))
        if ok:
            x, y = nx, ny
            q1, q2 = nq1, nq2
            print(f"XY=({x:.1f}, {y:.1f})")
        else:
            print("Timeout: no se movio")


def ajustar_escala_por_medicion(perfil):
    print("\nAjuste de escala XY por medicion real")
    print("Primero dibuja un cuadrado de prueba y mide ancho y alto en papel.")
    objetivo = float(input("Lado objetivo del cuadrado (mm), recomendado 80: ").strip() or "80")
    med_x = float(input("Ancho medido real en X (mm): ").strip())
    med_y = float(input("Alto medido real en Y (mm): ").strip())

    if med_x <= 0 or med_y <= 0:
        print("Valores invalidos")
        return

    kx = objetivo / med_x
    ky = objetivo / med_y
    perfil["workspace"]["escala_x"] *= kx
    perfil["workspace"]["escala_y"] *= ky

    print("Nueva escala:")
    print(f"escala_x={perfil['workspace']['escala_x']:.5f}")
    print(f"escala_y={perfil['workspace']['escala_y']:.5f}")


def ajustar_rotacion(perfil):
    print("\nAjuste de rotacion global")
    print("Si tus figuras salen inclinadas respecto a la hoja, corrige aqui.")
    delta = float(input("Grados a sumar (ejemplo 1.5 o -1.5): ").strip())
    perfil["workspace"]["rotacion_grados"] += delta
    print(f"rotacion_grados={perfil['workspace']['rotacion_grados']:.3f}")


def ajustar_suavidad_velocidad(perfil):
    print("\nAjuste de suavidad y velocidad")
    print("Mas pasos_dibujo = mejor recta/curva, pero mas lento.")
    print("Menos pasos_dibujo = mas rapido, puede perder definicion.")

    pasos_dibujo = input(f"pasos_dibujo actual {perfil['movimiento']['pasos_dibujo']}: ").strip()
    pasos_traslado = input(f"pasos_traslado actual {perfil['movimiento']['pasos_traslado']}: ").strip()
    timeout = input(f"timeout_movimiento actual {perfil['movimiento']['timeout_movimiento']}: ").strip()

    if pasos_dibujo:
        perfil["movimiento"]["pasos_dibujo"] = max(3, int(pasos_dibujo))
    if pasos_traslado:
        perfil["movimiento"]["pasos_traslado"] = max(2, int(pasos_traslado))
    if timeout:
        perfil["movimiento"]["timeout_movimiento"] = max(2.0, float(timeout))


def ajustar_lapiz(perfil):
    print("\nAjuste de asentamiento de lapiz")
    print("Sube un poco settle_down_s si ves relieve/manchas al apoyar.")

    down = input(f"settle_down_s actual {perfil['lapiz']['settle_down_s']}: ").strip()
    up = input(f"settle_up_s actual {perfil['lapiz']['settle_up_s']}: ").strip()

    if down:
        perfil["lapiz"]["settle_down_s"] = max(0.0, float(down))
    if up:
        perfil["lapiz"]["settle_up_s"] = max(0.0, float(up))


def imprimir_resumen(perfil):
    ws = perfil["workspace"]
    ang = perfil["angulos"]
    mov = perfil["movimiento"]
    lap = perfil["lapiz"]
    print("\n=== RESUMEN PERFIL ===")
    print(f"offset_x={ws['offset_x']:.3f}, offset_y={ws['offset_y']:.3f}")
    print(f"escala_x={ws['escala_x']:.6f}, escala_y={ws['escala_y']:.6f}")
    print(f"rotacion_grados={ws['rotacion_grados']:.3f}")
    print(f"q1_gain={ang['q1_gain']:.6f}, q2_gain={ang['q2_gain']:.6f}")
    print(f"q1_offset={ang['q1_offset']:.3f}, q2_offset={ang['q2_offset']:.3f}")
    print(f"pasos_dibujo={mov['pasos_dibujo']}, pasos_traslado={mov['pasos_traslado']}, timeout={mov['timeout_movimiento']}")
    print(f"settle_down_s={lap['settle_down_s']}, settle_up_s={lap['settle_up_s']}")


# ============================================================================
# MAIN
# ============================================================================
def main():
    perfil = cargar_perfil()
    print("Perfil cargado.")
    imprimir_resumen(perfil)

    try:
        arduino = serial.Serial(PUERTO_ARDUINO, BAUDRATE, timeout=1)
        time.sleep(2)
        while arduino.in_waiting > 0:
            print("Sistema:", arduino.readline().decode("utf-8", errors="ignore").strip())
        print("Conexion serial establecida.")
    except Exception as e:
        print(f"No se pudo abrir {PUERTO_ARDUINO}: {e}")
        sys.exit(1)

    try:
        while True:
            print("\n" + "=" * 44)
            print("CALIBRACION ROBOT SCARA")
            print("=" * 44)
            print("1) Modo jog y guardar centro de hoja")
            print("2) Dibujar cuadrado de prueba (80 mm)")
            print("3) Dibujar circulo de prueba (diametro 80 mm)")
            print("4) Ajustar escala XY por medicion")
            print("5) Ajustar rotacion global")
            print("6) Ajustar suavidad/velocidad")
            print("7) Ajustar asentamiento de lapiz")
            print("8) Ver resumen perfil")
            print("9) Guardar perfil")
            print("10) Solicitar paro seguro")
            print("11) Restaurar espacio de trabajo base")
            print("0) Home y salir")

            op = input("Selecciona opcion: ").strip()

            if op == "1":
                modo_jog_centro(arduino, perfil)
            elif op == "2":
                ok = trazar_polilinea(arduino, figura_cuadrado(80.0), perfil)
                print("OK" if ok else "Fallo en prueba")
            elif op == "3":
                ok = trazar_polilinea(arduino, figura_circulo(40.0, n=72), perfil)
                print("OK" if ok else "Fallo en prueba")
            elif op == "4":
                ajustar_escala_por_medicion(perfil)
            elif op == "5":
                ajustar_rotacion(perfil)
            elif op == "6":
                ajustar_suavidad_velocidad(perfil)
            elif op == "7":
                ajustar_lapiz(perfil)
            elif op == "8":
                imprimir_resumen(perfil)
            elif op == "9":
                guardar_perfil(perfil)
            elif op == "10":
                solicitar_paro()
                ir_home(arduino, float(perfil["movimiento"]["timeout_movimiento"]))
            elif op == "11":
                restablecer_espacio_trabajo(perfil)
            elif op == "0":
                ir_home(arduino, float(perfil["movimiento"]["timeout_movimiento"]))
                break
            else:
                print("Opcion invalida")

    except KeyboardInterrupt:
        print("Interrumpido por usuario.")
    finally:
        arduino.close()


if __name__ == "__main__":
    main()
